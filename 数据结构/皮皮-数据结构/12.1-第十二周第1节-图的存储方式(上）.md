图的存储结构：
	有向图/无向图：边集数组 邻接矩阵 邻接表
	有向图：十字链表
	无向图：多重邻接表


图的存储：主要就是存顶点数据和边
假设有n个结点，m条边

**顶点**数据：数组存顶点
**边**：边是由顶点表示的，存边的两个端点即可。
	一个边用一个结构体表示。

1、==边集数组==
	![[Pasted image 20260413115046.png]]
	无向边：
		struct Node {
			char x,y;//边
			int w;//边的权值
		}g\[];
	有向边：
		struct Node {
			char start,end;//边
			int w;//边的权值
		}g\[];

2、==邻接矩阵==:适用于==稠密图==，稀疏图会浪费空间。
	![[Pasted image 20260324204454.png]]
	![[Pasted image 20260324202417.png]]
	顶点数据：顶点数组存顶点
	v\[i] = 'A'.顶点数组中的下标就是点的编号
	边：二维数组
	-----
	无权有向图：二维数组来存边
	g\[i]\[j]=0. 没有i--->j的边
	g\[i]\[j]=1. 有i--->j的边
	-----
	带权有向图：取范围之外的值为无穷大，视作无权值（因为无法参与计算）。例如：边权不超过10000。inf=10001
	g\[i]\[j]=w. 有i--->j的边,边权为w的边
	g\[i]\[j]=inf(∞). 没有i--->j的边
	--------
	无权无向图：二维数组来存边
	要把一条无向边当成两条方向相反的有向边存
	g\[i]\[j]=g\[j]\[i]=0;
	存在一条i j之间的无向边
	g\[i]\[j]=g\[j]\[i]=1;
	-----
	带权无向图：要把一条无向边当成两条方向相反的 权值相等的有向边存
	g\[i]\[j]=g\[j]\[i]=inf;
	存在一条i j之间权为w的无向边
	g\[i]\[j]=g\[j]\[i]=w;

代码：
（==无权无向图==）：
```
#include <iostream>
using namespace std;

typedef char ElemType;

//=============================================
// 无权无向图 有 n(n<=100)个结点
//=============================================


// default =0; 邻接点 =1;
int graph[105][105];

// 图 的输入分为 图的所有的点的数据+图的所有的边 两部分
ElemType vertex[105];
int n;//顶点数
int m;//边数
ElemType x, y;// 两点表示一条线段
int xi, yi;// 两点下标

int Find(ElemType x)
{
	for(int i=1;i<=n;i++)
	{
		if (vertex[i] == x)
		{
			return i;
		}
	}
	// 没有该点的话
	return 0;
}


int main()
{
	// n个顶点vertex
	cin >> n;
	// m条边
	cin >> m;

	for (int i = 1; i <= n; i++)
	{
		cin >> vertex[i];
	}

	
	for (int i = 1; i <= m; i++)
	{
		cin >> x >> y;
		if (Find(x) && Find(y))
		{
			xi = Find(x);
			yi = Find(y);

			// 标记
			graph[xi][yi] = graph[yi][xi] = 1;
		}
	}

	// 输入一个点的数据 返回它的邻接点的个数
	char s; cin >> s;
	int si = Find(s);
	int res = 0;
	for (int i = 1; i <= n; i++)
	{
		if (graph[si][i] == 1 && si != i) res++;
	}
	cout << res << endl;

	return 0;
}

a c
```

```
//带权无向图为例。n个点 m条边  n<=100  0<=w<=10000

int Find(char x)
{
	int i=0;
	for(i=1;i<=n;i++)
	{
		if(v[i]==x)
		{
			break;
		}
	}

{
	scanf("%d %d",&n,&m);
	getchar();
	for(int i=1;i<=n;i++)
	{
		scanf("%c",&v[i]);
	}
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=n;j++)
		{
			g[i][j]=inf;
			//if(i==j)g[i][j]=0;
		}
	}
	char x,y;
	int xi,yi,w;

		getchar();

		yi=Find(y);

	int d=0;
```

3、邻接表：
顶点数据：顶点数组存顶点
v\[i]='A'.顶点数组中的下标就是点的编号

边：
	无向图：把每个点的邻接点 构成一个链表 挂在该点之后
	有向图：
		邻接表：把每个点的==出边==邻接点 构成一个链表 挂在该点之后
		==逆==邻接表：把每个点的==入边==邻接点 构成一个链表 挂在该点之后

代码;
```
#include<stdio.h>
#include<stdlib.h>
#include<string.h>
#define inf 10001
//带权无向图为例。n个点 m条边  n<100  0<=w<=10000
int n,m;
typedef struct ENode
{
	int adj;//邻接点的下标
	int w;//边权
	struct ENode* next;
}ENode;

struct Graph
{
	char data;//顶点数据
	ENode* first;//指向该点边链表（邻接点链表）的指针
}g[105];
int Find(char x)
{
	int i=0;
	for(i=1;i<=n;i++)
	{
		if(g[i].data==x)
		{
			break;
		}
	}
	return i;
}
int main()
{
	scanf("%d %d",&n,&m);
	getchar();
	for(int i=1;i<=n;i++)
	{
		scanf("%c",&g[i].data);
		g[i].first=NULL;
	}
	
	char x,y;
	int xi,yi,w;
	for(int i=1;i<=m;i++)
	{
		getchar();
		scanf("%c %c %d",&x,&y,&w);
		xi=Find(x);//找两个端点的编号
		yi=Find(y);
		//xi------->yi yi是xi的邻接点。yi 插入到xi到链表中
		ENode* e=(ENode*)malloc(sizeof(ENode));
		e->adj=yi;
		e->w=w;
		e->next=g[xi].first;
		g[xi].first=e;

		//yi------->xi xi是yi的邻接点。xi 插入到yi到链表中
		e=(ENode*)malloc(sizeof(ENode));
		e->adj=xi;
		e->w=w;
		e->next=g[yi].first;
		g[yi].first=e;
	}
	getchar();
	scanf("%c",&x);
	int d=0;
	xi=Find(x);
	ENode* p=g[xi].first;
	while(p!=NULL)
	{
		d++;
		p=p->next;
	}
	printf("%d\n",d);
	return 0;
}
/*
4 5
ABCD
A B 3
A D 6
A C 0
B D 9
D C -4
*/
```

自己写的：
```
	#include <iostream>

using namespace std;
typedef char ElemType;

//============================
// 带权有向图 的邻接表存储 
// 结点n(n<=100)个 
// 边m个
//============================

// 出边邻接点 链表的结点
typedef struct ENode
{
	int adj;// 出边邻接点的下标
	int w;// 出边的权值
	struct ENode* next;
}ENode;

// 图 的结点数组
struct Graph
{
	ElemType data;// 图结点的数据域
	ENode* first;// 图结点 的邻接点列表指针
}g[105];

int n, m;// n个点 m条边

int Find(ElemType x)
{
	for(int i = 1; i <= n; i++)
	{
		if (g[i].data == x) return i;
	}
	return 0;// 表示图不存在=x的点
}

int main()
{
	cin >> n >> m;
	for (int i = 1; i <= n; i++)
	{
		cin>>g[i].data;
		g[i].first = NULL;
	}
	// 因为不是按照程序运算顺序进行的输入，所以我们需要先保存用户的输入
	ElemType x, y;
	int weight,xi,yi;
	for (int i = 1; i <= m; i++)
	{
		// 用户输入: start end weight
		cin >> x >> y >> weight;

		xi = Find(x);
		yi = Find(y);
		// m个ENode
		ENode* newNode = (ENode*)malloc(sizeof(ENode));
		newNode->adj = yi;// 出边邻接点的下标
		newNode->w = weight;

		// 头插法加入到xi的链表中
		newNode->next = g[xi].first;
		g[xi].first = newNode;
	}

	// 检测
	ElemType s; cin >> s;
	int si = Find(s);
	int res = 0;
	ENode* p = g[si].first;
	while (p != NULL)
	{
		res++;
		p = p->next;
	}
	cout << res << endl;

	return 0;
}
/*
4 5
ABCD
A B 3
A D 6
A C 0
B D 9
D C -4
*/
```
