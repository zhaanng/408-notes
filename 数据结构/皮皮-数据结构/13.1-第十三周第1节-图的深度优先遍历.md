图的遍历：
深度优先遍历：
	从起点开始，每访问完一个点，就去找它的一个未被访问的邻接点，如果能找到就继续访问邻接点；如果找不到，就返回到上一步。------》基于==递归==（也就是==栈==）

如何理解深度？
	不能按照人为分层的层次理解深度
	深度：沿着走过的路径，离A越远，就越深。


遍历思路：
	（1）任选一个未被访问的点v做起点，访问该结点并且打标记。
	（2）找v的一个未被访问过的邻接点w。如果w存在，对w做同样的事（递归调用）。。。如果w不存在，说明v点到死胡同了，返回到上一个被访问的点。
	（3）此时如果图中还有未被访问过的点，再继续执行第一步


递归函数：DFS（i） //对点i进行深度优先访问
（1）访问i
（2）对i打标记
（3）找i的一个未被访问过的邻接点w，dfs(w)

#include<stdio.h>
#include<stdlib.h>
#include<string.h>

//有n（<=100）个点 m条边的无权无向图 对该图进行遍历。
//DFS时间复杂度： 邻接矩阵存图 O(n^2).   邻接表存图：O(n+m)
int n,m;
char v[105];//顶点数组
int g[105][105];//邻接矩阵 g[i][j]=g[j][i]=1 存在i----j
int flag[105];//标记某点有没有访问过
int Find(char x)
{
	int i=0;
	for(i=1;i<=n;i++)
	{
		if(v[i]==x)
		{
			break;
		}
	}
	return i;
}
//深度优先遍历
void DFS(int i)
{//对i点进行深度优先访问
	printf("%c ",v[i]);//访问
	flag[i]=1;//打标记 标记为已经访问的状态
	for(int j=1;j<=n;j++)
	{//找i的未被访问过的邻接点
		if(g[i][j]==1&&flag[j]==0)
		{
			DFS(j);
		}

	}
	
}
int main()
{
	scanf("%d %d",&n,&m);
	getchar();
	for(int i=1;i<=n;i++)
	{
		scanf("%c",&v[i]);
	}
	char x,y;
	int xi,yi;
	for(int i=1;i<=m;i++)
	{
		getchar();
		scanf("%c %c",&x,&y);
		xi=Find(x);
		yi=Find(y);
		g[xi][yi]=g[yi][xi]=1;
	}
	for(int i=1;i<=n;i++)
	{//考虑到非连通图
		if(flag[i]==0)
		{
			DFS(i);
		}
	}
	
	
	
	return 0;
}

#include <iostream>
using namespace std;

#define maxx 100
typedef char ElemType;

/*
==============================================
					无向图
==============================================
图的深度优先搜索 Depth First Search of Graph
	图的存储： 邻接矩阵
	思路：
		1、递归，从任意一个结点开始，
				访问这个点，访问他的 未被访问过的 邻接点，
				进入邻接点后，访问这个点，访问他的 未被访问过的 邻接点，
				直到访问的结点没有 未被访问过的 邻接点，
				返回上一级，继续访问该结点的 未被访问过的 邻接点，
				循环执行，
				直到返回到最顶层的循环，退出循环，
				函数结束。
*/
// 图的存储：邻接矩阵
ElemType v[maxx];
int g[maxx][maxx];// 初始化=0，存在邻接关系时=1
int flag[maxx];
int n,m;// n个结点 m条边
// 图的深度优先搜索
void DFS(int i)
{
	cout << v[i];// 访问操作
	flag[i] = 1;// 标记i点状态
	// 遍历i所有邻接点
	for (int j = 1; j <= n; j++)
	{
		// i和j邻接 && j未被访问过
		if (g[i][j] == 1 && flag[j] == 0)
		{
			DFS(j);
		}
	}
	return;
}
// 依据数据x查找在v[]数组的下标
int Find(ElemType x)
{
	for (int i = 1; i <= n; i++)
	{
		if (v[i] == x)
			return i;
	}
	return -1;// 如果不存在，返回错误 index
}

int main()
{
	// 邻接矩阵 存储图 :点 边
	cin >> n >> m;
	for (int i = 1; i <= n; i++)
	{
		cin >> v[i];
	}
	char x, y;
	int xi, yi;
	for (int i = 1; i <= m; i++)
	{
		cin >> x >> y;
		xi = Find(x);
		yi = Find(y);
		g[xi][yi] = 1;
		g[yi][xi] = 1;
	}
	DFS(1);

	return 0;
}

#include <iostream>
using namespace std;
#define maxx 105
typedef char ElemType;

/*
无向图 非递归 栈实现
图的深度优先遍历 
	图的存储：邻接矩阵
	思路:
		用栈模拟递归过程，
		入栈 起点，
		while循环 检查 栈非空，
			出栈 元素，
			访问该元素，
			倒序，把邻接点入栈，
	直到while循环结束
*/
//==============================栈============================
// 顺序栈 结构体
typedef struct satck
{
	ElemType data[maxx];
	int top;// 栈顶指针
}stack;
// 栈初始化
stack* InitStack()
{
	stack* s = (stack*)malloc(sizeof(stack));
	s->top = 0;//表示指向的位置 也表示栈内元素的个数

	return s;
}
// 压栈 :栈顶指针 指向 栈顶元素
void PushStack(stack* s,ElemType e)
{
	// 判断栈满
	if (s->top == maxx-1)
	{
		cout << "栈已满，无法压栈" << endl;
		return;
	}
	s->data[++s->top] = e;
}
// 栈判空
bool isEmpty(stack* s)
{
	if (s->top == 0)
		return true;//空
	else
		return false;//非空
}
// 出栈
ElemType PopStack(stack* s)
{
	// 判栈空
	if (isEmpty(s))
	{
		cout << "栈为空，无法出栈" << endl;
		return '0';
	}
	ElemType res = s->data[s->top];
	s->top--;
	return res;
}
//====================================================================
//===============================邻接矩阵=============================
int n, m;// n个数 m条边
ElemType v[maxx];// 存 结点数据 的数组
int g[maxx][maxx];// 存 点之间关系，也就是 存边 的数组
int flag[maxx];// 存 点状态的数组 ：是否遍历到
//====================================================================
//==========================深度优先遍历 DFS==========================
int Find(ElemType x)
{
	for (int i = 1; i <= n; i++)
	{
		if (v[i] == x)
			return i;
	}
	return -1;// 不存在则返回 错误下标值
}
// DFS 栈模拟递归 实现 图的深度优先遍历
void DFS(int x)
{
	// 压入遍历起点
	stack* s = InitStack();
	PushStack(s,v[x]);
	// 循环
	ElemType e;
	int ei;//元素在v[]下标
	while (!isEmpty(s))
	{
		// 出栈
		e=PopStack(s);
		ei = Find(e);
		// 访问 未被访问过的点
		if (flag[ei] == 1)continue;//跳过本次循环
		cout << e;
		// 标记被访问的元素
		flag[ei] = 1;
		// 所有邻接点 倒序 入栈
		for (int i = n; i >= 1; i--)//倒序查找
			if (g[ei][i] == 1 && flag[i] == 0)
				PushStack(s, v[i]);// 所有符合条件的邻接点的数据入栈
	}
}
int main()
{
	cin >> n >> m;
	// 存n个点
	for (int i = 1; i <= n; i++)
	{
		cin >> v[i];
	}
	// 存m条边
	ElemType x, y;
	int xi, yi;
	for (int i = 1; i <= m; i++)
	{
		cin >> x >> y;
		xi = Find(x);
		yi = Find(y);
		g[xi][yi] = g[yi][xi] = 1;
	}

	DFS(1);

	return 0;
}
