==什么是树的遍历？==  ——》  \[==IMPORTANT==]
	那么对于遍历一棵树来说，循环过程就是，找到一个结点的最左最下的左子树为空的树结点，然后判断是否有右子树，有则跑到右子树里继续循环。不同的遍历只是访问结点的时机不同。

这样，线索化的==目的==就是：
	给原先需要循环往复的找到 一个点的遍历的下一个结点 的过程，加了一个**传送门**，当到一个结点时，如果发现有传送门，可以坐传送门直接到达 正确的下一个结点，如果没有，就按部就班的 去找。

二叉树中的结点按照某种规则进行遍历，会产生一串有一定顺序的结点链，这叫二叉树的遍历。

=================================================
我们如何在存储非线性结构的二叉树时可以==线性访问==二叉树呢？

思路是：在链表结点中再加两个地址域来保存前驱和后继。保存地址8Byte+8Byte
	把某种遍历顺序保存起来，比如 先序遍历  中序遍历 后序遍历
=================================================
但这样在数据量很大时很==占用空间==，如何减小空间的占用呢？

我们发现保存lchild/rchild的结点一定不会都存满，所以我们可以充分利用剩余的空间，只需要在l/rchild前后添加2个int flag进行说明4Byte+4Byte，说明是前驱后继/lchild rchild，这样每个结点可以节省8Byte
	链中某结点的前一个结点叫做他的前驱，后一个叫做后继。
	如果二叉树在存储时是用每个结点除存 数据 外还存 lchild 和 rchild指针。
	这时候，设一共有n个结点，可知一共会有2n个存放child指针的空间。
	而==child空间一定填不满==，因为总共n个结点中，一共有n-1个结点可以做child，且每个结点只可以做一个结点的child，所以一定会有n-1个child，所以==一定会剩下n+1个child空间被浪费==。
	我们可以将这部分空间利用起来。
	在这里，如果结点有空的child空间，我们可以将结点的前驱/后继存放到里面。
	有两个child的结点没法用这种方法存储其前驱/后继，我们要用其他方法，最终要实现全部结点都要储存其前驱/后继。
	==二叉树是非线性结构，无法按照线性结构的前后循序进行遍历读取==，我们就按照一定规则，找到每个结点的前驱/后继，最终实现非线性结构的线性读取，这叫做==线索化==。
	==注意==： 前驱/后继/lchild/rchild保存的都是指针地址，所以需要加一个flag进行说明

	具体操作如何操作？
		中序遍历操作时，将前驱后继内容添加到结点中，**因此在储存二叉树时，要先遍历一遍**。
		然后结点没有前驱的指针，**找不到前驱**，所以需要一个pre指针，指向结点的前驱主要是前驱，不是父亲结点。
		访问某结点时，可以存储其前驱地址，但**无法确认后继地址**，因此需要当访问其后继结点时才可以存储当前结点的后继。
		因此访问该结点时，可以存储 **该结点前驱** 和 **该结点前驱的后继**。

```c
#include<stdio.h>
#include<stdlib.h>
//二叉链表的结点结构
typedef struct BTNode{
    char data;
    struct BTNode* l;
    struct BTNode* r;
    int lflag,rflag;//flag==0 指针域指向孩子。 ==1 指向线索
}BTNode,*BTree;
BTNode* pre=NULL;
BTree InitBTree(char root)
{
    BTNode *s=(BTNode*)malloc(sizeof(BTNode));
    s->data=root;
    s->l=s->r=NULL;
    s->lflag=s->rflag=0;
    return s;
}
BTNode* Find(BTree ro,char fx)
{
    if(ro->data==fx)
    {
        return ro;
    }
    BTNode* ans=NULL;
    if(ro->l!=NULL&&ro->lflag==0)//注意细节
    {
        ans=Find(ro->l,fx);
        if(ans!=NULL)return ans;
    }
    if(ro->r!=NULL&&ro->rflag==0)
    {
        ans=Find(ro->r,fx);
        if(ans!=NULL)return ans;
    }
    return NULL;
}
BTree Insert(BTree ro,char x,char fx,int flag)
{
    BTNode* f=Find(ro,fx);
    BTNode* s=(BTNode*)malloc(sizeof(BTNode));
    s->data=x;
    s->l=s->r=NULL;
    s->lflag=s->rflag=0;
    if(flag==0)f->l=s;
    else f->r=s;
    return ro;

}
void Visit(BTNode* p)
{//p和pre就是一对前驱和后继关系
    //正在访问p
    if(p->l==NULL)
    {//给p添加前驱线索
        p->l=pre;
        p->lflag=1;
    }
    if(pre!=NULL&&pre->r==NULL)//注意细节
    {//给pre添加后继线索
        pre->r=p;
        pre->rflag=1;
    }
    pre=p;

}
void InOrderBTree(BTree ro)
{//中序遍历
    if(ro==NULL)return;//递归出口

    InOrderBTree(ro->l);

    Visit(ro);//添加线索

    InOrderBTree(ro->r);

}
int main()
{
    int n;
    int flag;//flag==0 左孩子。==1 右孩子
    scanf("%d",&n);
    getchar();
    char root,x,fx;
    scanf("%c",&root);
    BTree ro=InitBTree(root);
    for(int i=1;i<=n-1;i++)
    {
        getchar();
        scanf("%c %c %d",&x,&fx,&flag);
        ro=Insert(ro,x,fx,flag);
    }
    
    //进行线索化：基于中序遍历添加线索
    InOrderBTree(ro);

    //找x 在中序遍历序列中的前驱和后继
    getchar();
    scanf("%c",&x);
    
    BTNode* p=Find(ro,x);

    //找前驱
    if(p->l==NULL)printf("无前驱\n");
    else
    {
        if(p->lflag==1)printf("前驱是%c\n",p->l->data);
        else
        {//p->l指向的是左孩子 不是前驱 并且p->l不为空 此时前驱一定在p的左子树中 而且一定是左子树中最靠右的结点
            BTNode* q=p->l;
            //q 只要是右孩子存在 就往右边走
            while(q->r!=NULL&&q->rflag==0)q=q->r;
            printf("前驱是%c\n",q->data);
        }


    }
    //找后继
    if(p->r==NULL)printf("无后继\n");
    else
    {
         if(p->rflag==1)printf("后继是%c\n",p->r->data);
         else
         {//p->r指向的是右孩子 不是后继 并且p->r不为空 此时后继一定在p的右子树中 而且一定是右子树中最靠左的结点
             BTNode* q=p->r;
            //q 只要是左孩子存在 就往左边走
            while(q->l!=NULL&&q->lflag==0)q=q->l;
            printf("后继是%c\n",q->data);
         }
    }

    return 0;
}
```

自己写的代码：
```c++
#include <iostream>
using namespace std;
typedef char ElemType;
#define LEFT_CHILD 0 // 左孩子
#define RIGHT_CHILD 1 // 右孩子

// 二叉链表 做 树的结点
typedef struct BTNode
{
    ElemType data;// 树结点保存的数据
    struct BTNode* l;// 左孩子指针
    struct BTNode* r;// 右孩子指针
    // 充分利用空闲指针——线索化
    int lflag, rflag;// ==0 指向 左右孩子，==1时 已经指向 遍历前驱，表示指针状态
}BTNode,*BTree;
BTNode* pre=nullptr;// 后面会用到,保存前驱结点
// 初始化 二叉树结点
BTNode* CreateBTNode(ElemType data)
{
    BTNode* node = (BTNode*)malloc(sizeof(BTNode));
    node->data = data;
    node->l = node->r = nullptr;
    node->lflag = node->rflag = 0;

    return node;
}
#define InitBTree CreateBTNode // 将初始化树根结点的函数单独命名
// Find() 依据data，找到root上数据为data的结点的地址，并返回
BTNode* Find(BTree root, ElemType data)
{// 递归 先序遍历 查找
    if (root->data == data)
        return root;
    BTNode* res = nullptr;
    if (root->l != nullptr&&root->lflag==0)// 保证左孩子指针指向的是左孩子
    {
        res = Find(root->l,data);
        if (res != nullptr) return res;
    }
    if (root->r != nullptr&&root->rflag==0)
    {
        res = Find(root->r, data);
        if (res != nullptr) return res;
    }
    return nullptr;
}
// 插入 树结点
// 要插入的树 BTree root，要插入的结点的数据 ElemType data，
// 要插入的结点的父亲结点的数据（确定插入位置） ElemType fa
// 做左子树还是右子树 int flag, LEFT_CHILD, RIGHT_CHILD
void Insert(BTree root, ElemType data, ElemType fa, int flag)
{
    BTNode* newNode=CreateBTNode(data);// 创建一个树结点 newNode
    BTNode* pos = Find(root,fa);// 找到要插入的地方
    if (flag == LEFT_CHILD)
        pos->l = newNode;
    else
        pos->r = newNode;
}
// Visit()操作结点,这里是 打上线索
void Visit(BTNode* cur)
{
    // 检查cur结点地左孩子是否可以线索化,存前驱
    if (cur->l == nullptr)// 可以存
    {
        cur->l = pre;// 前驱是 pre
        cur->lflag = 1;
    }
    // 检查是否可以存后继,存前一个结点pre的后继，记得更新pre=cur
    
    // 因为我们无法得知cur结点地后继是啥，所以没法存cur的后继
    // 但可以得知pre的后继是cur，
    // 所以结点的后继只能访问到结点的下一个结点时才能存取
    if (pre != nullptr && pre->r == nullptr)// 可以存
    {
        pre->r = cur;
        pre->rflag = 1;
    }
    pre = cur;// 访问完cur后，更新pre
}

// 递归 遍历 中序遍历 
void InOrderBTree(BTree root)
{
    if (root == nullptr) return;

    if(root->lflag==0)
        InOrderBTree(root->l);

    Visit(root);

    if (root->rflag == 0)
        InOrderBTree(root->r);
}

int main()
{
    int n, flag;// n 个结点 flag 插入做左/右孩子
    cin >> n;
    ElemType data;// 结点数据
    cin >> data;
    BTree root=InitBTree(data);// 创建二叉树
    ElemType x, fx;

    for (int i = 1; i <= n-1; i++)// 构建树
    {
        cin >> x >> fx >> flag;
        Insert(root, x, fx, flag);// 插入树结点
    }

    //========树 已经构建完成========

    // 进行线索化——》遍历，同时线索化
    InOrderBTree(root);
    
    //=======树 已经完成线索化========

    // 运用线索化的树 找结点x在中序遍历下的前驱/后继
    ElemType tar; cin >> tar;// 要找的结点
    BTNode* p = Find(root, tar);

    // 找前驱
    // 1、用线索化直接保存的前驱
    // 2、没有线索化，按部就班的找，中序遍历中 结点的前驱是其左子树的最右端的结点
    if (p->l == nullptr) cout << p->data<<" 无前驱" << endl;
    else if (p->lflag == 1)// 情况1
            cout << "线索化查找， " << p->data << " 的前驱是 " << p->l->data << endl;
    else// 情况2
    {
        BTNode* q = p->l;// 换到左子树
        // 找到最右端的孩子结点
        
        // 两种情况 
        // 1、一直循环，如果遇到rflag==1，说明 当前指向的p就是最右端的结点
        // 2、一直循环，默认rflag==1，直到p->r==nullptr，说明 当前指向的p就是最右端的结点
        while (q->r != nullptr&&q->rflag!=1)
        {
            q = q->r;
        }
        cout << "普通查找前驱， " << p->data << " 的前驱是 " << q->data << endl;
    }

    // 找后继
    // 和前驱一样
    // 只不过后继的 普通查找结点中序遍历后继是找，结点的右子树的最左端的结点
    if (p->r == nullptr) cout << p->data << " 无后继" << endl;
    else if (p->rflag == 1)// 情况1
        cout << "线索化查找， " << p->data << " 的后继是 " << p->r->data << endl;
    else// 情况2
    {
        BTNode* q = p->r;
        
        while (q->l != nullptr && q->lflag ==0)
        {
            q = q->l;
        }
        cout << "普通查找后继， " << p->data << " 的后继是 " << q->data << endl;
    }
    

    return 0;
}
```
