最短路径问题：
BFS——无权图的单源最短路径问题

单源最短路径问题：起点固定，求单一起点到其他点的最短距离。
Dijkstra算法——O(n^2)——》不能有负数边权

多源最短路径问题：求任意两点之间的最短距离。
Floyd算法——》基于动态规划求解

1、最优子结构性质 重叠子问题性质

D算法的==关键思想==是：
Dijkstra算法：if(dist\[k]+a\[k]\[j]<dist\[j])——》 dist\[j]=dist\[k]+a\[k]\[j]
==尝试==让任意两点之间距离变短的方法：在两点之间引入中转点

==引入的中转点越多，变短的可能性越大==

假设带权图G 含有n个点 m条边，在n个都允许做中转点的情况下，求任意两点(i,j)之间的最短距离？
采用==邻接矩阵==存图 a\[i]\[j]——》这样在给dp数组赋值时方便

2、找状态数组

在n个都允许做中转点的情况下，求任意两点(i,j)之间的最短距离？
——》在n个都允许做中转点的情况下，求任意两点(i,j)之间的最短距离ans
dp\[n]\[i]\[j]=ans

3、确定初始状态 中间状态 最终状态

初始状态：不允许任何点做中转点 dp\[0]\[i]\[j]=a\[i]\[j]

中间状态：dp\[k]\[i]\[j]：允许用前k个点做中转点的情况下，任意两点(i,j)之间的最短距离

最终状态：dp\[n]\[i]\[j]：在n个点都允许做中转点的情况下，任意两点(i,j)之间的最短距离 ans

4、状态转移方程

求中间状态方程：dp\[k]\[i]\[j]：允许用前k个点做中转点的情况下，任意两点(i,j)之间的最短距离？
假设dp\[0]\[i]\[j]\~\~\~dp\[k-1]\[i]\[j]都已经知道
==dp\[k]\[i]\[j]=min(dp\[k-1]\[i]\[j]),dp\[k-1]\[i]\[k]+dp\[k-1]\[k]\[j])==
走k的话，就成了 i到k的最短距离+k到j的最短距离

5、写代码

我们发现，Floyd算法的
**时间复杂度O(n^3)
空间复杂度O(n^3)**
空间复杂度太大，无法处理大量数据，需要减小空间开支，
降维，将三维数组降为二维数组。

==降维==：

能降维，因为每次dy都是为了增大k值，直到k\==n
而我们发现，k值的大小取决于k-1的取值大小，和更之前的k取值无关，
所以可以在保存k-1的数组上直接覆盖更新k的取值，
注意要用到的k-1别先盖掉就行
注意：降维，只节省空间，**任何变量的含义都不会变**

三层循环的顺序不能改变，**最外层枚举k**

```c
#include<stdio.h>
#include<stdlib.h>
#define inf 10005
int minn(int p,int q){return p<q?p:q;}
int n,m;//n<=100,边权不超过10000,假设n个点就是1～～n,以有向图为例
int a[105][105];
int dp[105][105];//dp[i][j]: 允许用前k个点做中转点的情况下，任意两点(i,j)之间的最短距离
//时间复杂度：O(n^2) 空间复杂度O(n^2)
int main()
{
	scanf("%d %d",&n,&m);
	//初始化邻接矩阵
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=n;j++)
		{
			a[i][j]=inf;
			if(i==j)a[i][j]=0;//自己到自己的最短距离是0
		}
	}
	int x,y,w;
	for(int i=1;i<=m;i++)
	{
		scanf("%d %d %d",&x,&y,&w);
		a[x][y]=w;
	}
	//初始状态
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=n;j++)
		{
			dp[i][j]=a[i][j];//初始时 不允许经过中转点，两点间的最短距离就是直连的边
		}
	}
	//求中间状态 最终得到最终状态
	// 降维
	for(int k=1;k<=n;k++)//枚举中转点，现在把第k个加入做中转点
	{
		for(int i=1;i<=n;i++)
		{
			for(int j=1;j<=n;j++)
			{
			 dp[i][j]=minn(dp[i][j], dp[i][k]+dp[k][j]);
			}
		}
	}
	
	//dp[i][j]就是答案
	for(int i=1;i<=n;i++)
	{
		for(int j=1;j<=n;j++)
		{
			printf("%d--->%d 的最短距离是:%d\n",i,j,dp[i][j]);
		}
	}
	return 0;
}
```

```c++
#include <iostream>
using namespace std;
#define maxx 105
#define inf 10005

int n, m;// n个点,m条边，n<=100
int a[maxx][maxx];// 邻接矩阵存图
int dp[maxx][maxx];// 降维，状态数组由三维降到二维

int main()
{
    cin >> n >> m;
    // 初始化邻接矩阵
    // 默认不同点之间的距离为inf
    // 自身到自身的距离为0
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            a[i][j] = inf;
            if (i == j)a[i][j] = 0;
        }
    }
    // 存边，更新
    int x, y, w;// x点到y点 权值为w 的有向边
    for (int i = 1; i <= m; i++)// 共m条边
    {
        cin >> x >> y >> w;
        a[x][y] = w;
    }
    // 初始化状态数组
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            dp[i][j] = a[i][j];
        }
    }
    // 状态转移方程
    // 能降维，因为每次dy都是为了增大k值，直到k==n
    // 而我们发现，k值的大小取决于k-1的取值大小，和更之前的k取值无关，
    // 所以可以在保存k-1的数组上直接覆盖更新k的取值，
    // 注意要用到的k-1别先盖掉就行
    // 注意：降维，只节省空间，任何变量的含义都不会变
    for (int k = 1; k <= n; k++)
    {
        for (int i = 1; i <= n; i++)
        {
            for (int j = 1; j <= n; j++)
            {
                dp[i][j] = min(dp[i][j], dp[i][k] + dp[k][j]);// 等号两边的dp[i][j]不相等
            }
        }
    }
    // 最终的dp[][]就是k==n时的状态
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            if (dp[i][j] == inf)cout << i << "点无法到达" << j << "点" << endl;
            else cout << i << "点到" << j << "点的距离是"<<dp[i][j] << endl;
        }
    }

    return 0;
}
```
