```c
AVLTree Delete(AVLTree root,int k)
{
    if(root==NULL)
    {
        printf("该数据不存在，无法删除\n");
        return root;
    }
    if(root->data==k)
    {//删除root
        if(root->l!=NULL&&root->r!=NULL)
        {//root度为2
            //找root度前驱
            AVLNode* p=root->l;
            while(p->r!=NULL)p=p->r;
            //p就是root的前驱
            root->data=p->data;
            root->l=Delete(root->l,p->data);
            //在root的左子树中删除了一个结点，可能导致root左右子树高度差扩大到2，root可能失衡
            //可能出现RR/RL失衡
            if(Geth(root->r)-Geth(root->l)>1)
            {
                AVLNode* right=root->r;
                if(Geth(right->r)>=Geth(right->l))
                {//RR
                    root=RRrotation(root); 
                }
                else
                {//RL
                    root=RLrotation(root); 
                }
            }

        }
        else{
            //度为1/0
            AVLNode* p=root;//p指向要删除的结点
            if(root->l!=NULL)root=root->l;
            else root=root->r;
            free(p);
            p=NULL;
        }
    }
    else if(k<root->data)
    {
        root->l=Delete(root->l,k);
        //在root的左子树中删除了一个结点，可能导致root左右子树高度差扩大到2，root可能失衡
        //可能出现RR/RL失衡
        if(Geth(root->r)-Geth(root->l)>1)
        {
                AVLNode* right=root->r;
                if(Geth(right->r)>=Geth(right->l))
                {//RR
                    root=RRrotation(root); 
                }
                else
                {//RL
                    root=RLrotation(root); 
                }
        }

    }
    else
    {
        root->r=Delete(root->r,k);
        //在root的右子树中删除了一个结点，可能导致root左右子树高度差扩大到2，root可能失衡
        //可能出现LL/LR失衡

                    root=LLrotation(root);

                    root=LRrotation(root); 
                }
        }
    }

}
```

算法：并查集 哈夫曼树

==并查集==：
	基于树形结构，处理==**不相交**==**集合的合并和查询**的问题。

我们有一些数据集合，对于这些集合存在两个操作：
1：给出两个数据x，y，把x所在的集合和y所在的集合合并
2：给出两个数据x，y，问两数据是否在同一个集合？

======================================================
给出n个数据，初始时每个数据自成一个集合。
给出若干操作序列：
op x y
1 4 6  第一种操作，4 集合， 6 集合
1 7 2  。。。
2 4 7 第二种操作， 4 数字， 7 数字

操作一：把两集合合并
操作二：检查两数是否在同一集合

2 1 4 6 7 3
输出：
![[Pasted image 20260519194936.png]]
NO

========================================

上面提出的两种操作，

首先可以用 **==数组==** 来解决。
	多个一维数组的集合，就是 **二维数组**
	先按照**次序**，存储所有一维数组
	操作一，**合并** a b数组，
		找到a b数组，a 所在行不变，b 所在行全部移动到 a 的所在行上，a 行清零，
		合并完成
	操作二，**查询**x y 两数是否在同一集合？
		找到x y，x 所在行数 ix，y所在行数 iy，
		比较 ix\==iy ? ture : false ;
		检查完成

再可以用 **树** 来解决。
	用树存储集合。
	操作一，**合并**：
		就是一棵树的根结点做另一棵树的根结点的父亲。即一棵树成为另一棵树的子树。
		==**优化**：一般是**矮的树合并到高的树**中，矮的成子树。==（因为高的做子树的话，会使新的树的高度是 **max(h1,h2)+1**。矮的做子树，新树高度是 max(h1,h2)。）
	操作二，**查询**：同一个集合——》同一棵树
		如何查两个数据是否在同一棵树中？
			找到两个数据的根结点，如果根结点相同，则属于同一棵树，即同一个集合。
		而且这种方法效率很高，最多查询树的高度次logn次。
	这种集合就叫做 **并查集**。

用树进行存储，实现合并和查询操作的集合——》并查集

普通的树的存储方法：
	双亲表示法
	孩子
	孩子兄弟

这里因为要向上查询找到根结点，所以需要用 **双亲表示法**

	1、递归
		==路径压缩==
			想办法把所有的子树都连接到根上，
			这样可以减小查询的总次数，提高查询效率

并查集代码：
	这个代码，实现了路径压缩，第一次Find()后，后续的Find()的效率极大提高，

```c
#include<stdio.h>
#include<stdlib.h>

// 每个数据范围在 0~100
int fa[105];//n<=100;
int n,m;//初始有n个数据1~~n m个操作
int op,x,y;//操作数op  

// Find() 找结点x所在树的根结点数据f
int Find(int x)
{
	if(fa[x]==-1)
	{
		return x;
	}
	else
	{
		int f=Find(fa[x]);//父亲的根就是x的根
		fa[x]=f;//路径压缩，将结点直接连到根上。
		return f;
		//return fa[x]=Find(fa[x]);
	}
}
int Find1(int x)
{
	int p=x;
	while(fa[p]!=-1)
	{
		p=fa[p];//引以为戒
	}
	return p;
}
int main()
{
	scanf("%d %d",&n,&m);
	for(int i=0;i<=n;i++)
	{
		fa[i]=-1;//-1不存在
	}
	int fx,fy;
	for(int i=1;i<=m;i++)
	{
		scanf("%d %d %d",&op,&x,&y);
		if(op==1)
		{//合并
			//找到x和y的根结点
			fx=Find(x);
			fy=Find(y);
			fa[fy]=fx;//fa[fx]=fy;
		}
		else
		{//查询
			//找到x和y的根结点
			fx=Find(x);
			fy=Find(y);
			if(fx==fy)printf("YES\n");
			else printf("NO\n");
		}
	}
	
	
	return 0;
}
```

```c++
#include <iostream>
#define maxx 105
using namespace std;

/*
    并查集 Union-Find
    对树的结点进行的两个操作:
    合并 查询
    1、合并：合并两棵树，普通的树，多叉树
        找到两棵树的根结点，将一棵的根结点接到另一棵的根结点上，
        即一棵的根节点成了另一棵的根结点的父亲结点
    2、查询：给出两个结点，查询这两棵树是否在同一棵树上
        找到两棵树的根结点，检查是不是同一个，
        如果是，返回 YES
        如果不是，返回 NO

    总之，就是找到两棵树的根结点，然后再进一步处理。
*/

int fa[maxx];// 父亲结点数组，数组值是子结点的父亲结点的，数组下标是该子结点的数值

// 找数据x所在树的根结点的数值,返回根结点的数值
// 迭代版
int Find(int x)
{
    int p = x;
    while (fa[p] != -1)// 根结点的fa[]==-1
        p = fa[p];
    return p;// 返回根结点数值
}
// 递归版
int Find2(int x)
{
    // 终止条件
    if (fa[x] == -1)// 此时x是根结点
    {
        return x;
    }
    else
    {
        int root = Find2(fa[x]);
        // 路径压缩
        // 把x变成root的子结点，这样第一次Find()完后，
        // 将两棵树都变成的2层的树，后续的Find()效率会非常高。
        fa[x] = root;
        return root;
        // return fa[x]=Find(fa[x]);
    }
}
int main()
{
    // n棵一个结点的树 后续要m次操作
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < maxx; i++)
        fa[i] = -1;
    // m次操作
    for (int i = 1; i <= m; i++)
    {
        int op, x, y;// op哪种操作 1 2，x y要操作的两个结点的数据
        cin >> op >> x >> y;
        if (op == 1)// op1 合并操作
        {
            // 找到结点的树的根结点的数据
            int rx, ry;
            rx = Find2(x);// 递归版
            ry = Find2(y);
            fa[rx] = ry;// 把树rx合并为树ry的子树
            // 这里可以优化，把矮树加到高树上，新树的高度为 h==max(h1,h2)/max(h1,h2)+1
        }
        else // op2 查询操作
        {
            int rx, ry;
            rx = Find(x);// 迭代版
            ry = Find(y);
            if (rx == ry)
                cout << "YES" << endl;
            else
                cout << "NO" << endl;
        }
    }
    
    return 0;
}
```
