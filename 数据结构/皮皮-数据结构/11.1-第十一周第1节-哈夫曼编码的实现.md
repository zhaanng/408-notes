#include <iostream>
#include <string.h>
#include <stdlib.h>

#define	MAXSIZE	100

using namespace std;


/*
	创建哈夫曼树HuffmanTree，
		树生成 思路：
			哈夫曼树是二叉树的一种，是树的带权路径长度最小的树，又称为最优二叉树。
			它在逻辑上的构建过程是： {
				有n个结点，每个结点有对应权值，每个结点看作一棵树，树的权值是根结点的权值。
				比较所有结点的权值，权值最小的两棵树合并，形成一个新的结点，是它们的父亲结点，父亲结点的权值是两子结点的和，父亲结点同时也是新树的根结点。
				新生成的树作为新结点加入到比较中，重复过程，最终只剩一棵树，结束。
			}；
		现象/特征：
				在保证树的带权路径长度最小的前提下，
				1、原先的n个结点全部会成为新树的根结点，且只有原先的n个结点做根结点。
				2、因此会产生新的n-1个结点。因此n个结点生成的哈夫曼树一共有2n-1个结点。
				3、根结点的权值越大，从树根到它的路径就越短。
				4、给从父亲结点到子结点的方向做个编码，往左是1，往右是0；
					再找从根结点到叶子结点的路径，叶子结点的权值越大，路径编码越短。
				5、根结点到叶子结点的路径在树上，唯一。
					即 给出一串哈夫曼树上生成的路径编码，可以识别是在树上以什么路径走的，路径编码分别是到达哪个叶子结点的路径。

*/

/*
	生成HuffmanTree哈夫曼树：
		树的结点的存储，
			w 通过权重决定该结点合并的优先级。所以需要保存每个结点的 权重。
			fa 我们需要通过检查 该结点的父亲结点是否存在 来判断 该结点是否进行了合并 。每个结点只需要合并一次。因此我们需要保存每个结点的 父亲结点的下标。
			lch,rch 在创建好HuffmanTree后，我们需要通过 从根结点访问到目标叶子结点来该获取该结点的在该树上的路径编码，因此需要保存每个结点的 左孩子下标 和 右孩子下标。

*/
typedef struct Node {
	int w;	// 结点权重
	int fa;	// 结点父亲结点下标
	int lch, rch;	// 结点左、右孩子结点下标

}HuffmanNode;

/*
	创建HuffmanTree
		根据输入的 n个要编码的结点可知，HuffmanTree 一共会有 2n-1个结点，因此需要手动开辟 2n-1个结点的空间。

		过程；
			根据逻辑上创建的过程，存储好 2n-1个结点后，
			我们需要在每次循环中，在 当前所有可以合并的树中（即fa == -1）找到权值最小的两棵树，
			对他俩进行合并，
			重复这个过程，直到只剩下一棵树，即HuffmanTree

	参数：要编码的结点数int n，每个结点的权值（结点权值的数组）int we[]
	返回值：返回保存结点数组的指针 HuffmanNode *
*/

/*
	void Find(HuffmanNode* tree, int n, int &m1, int &m2);
		在数组某一个区间里，找到权值最小的两个结点的下标，保存到传入的两个盒子里
	参数:tree 要查找到树；n 要查找的范围（随着合并的进行，数组里保存的结点会增加）；
		 m1 保存权值最小的结点的下标；m2 保存权值次小的结点的下标
	思路：
		依次遍历访问数组，
		给装最小值minn的盒子一个初始值（遍历数组，找一个符合条件的结点 赋值）
			在找最小值时，初始化一个没有父亲结点且没出界的结点即可；
			在找次小值时，初始化一个非最小值且没有父亲结点且没出界的结点；
		当访问结点 没有父亲结点 而且 访问没有过界 时，
		比较访问结点与minn盒子里的值的大小，刷新minn里的值，
		遍历完后得到最小值，再遍历一次，得到次小值，
		复制到m1 m2里

*/
void Find(HuffmanNode* tree, int n, int& m1, int& m2) {
	// 创建装最小值minn的盒子 权值最小的结点的下标
	int minn = -1;
	// 初始化minn，用来找最小值
	for (int i = 0; i < n; i++) {
		if (tree[i].fa == -1) {
			minn = i;
			break;
		}
	}
	// 遍历找最小值
	for (int i = 0; i < n; i++) {
		if (tree[i].w < tree[minn].w && tree[i].fa == -1) {
			minn = i;
		}
	}

	// 最小值传给m1
	m1 = minn;

	// 初始化minn，用来找次小值
	for (int i = 0; i < n; i++) {
		/*
		tree[i].w != tree[m1].w
			这里用权值来判断 最小值和 次小值的不能取同一个字符的方法是错误的
			因为当存在权值相等的字符时，这种方法就会出错，而权值相等的字符很常见。
			所以正确的办法是，
			用下标来确定是不是最小值所在的位置
		*/
		if (/*tree[i].w != tree[m1].w*/ i != m1 && tree[i].fa == -1) {
			minn = i;
			break;
		}
	}

	// 遍历找次小值
	for (int i = 0; i < n; i++) {
		if (tree[i].fa == -1 && tree[i].w < tree[minn].w && tree[i].w != tree[m1].w) {
			minn = i;
		}
	}

	// 次小值传给m2
	m2 = minn;

}

HuffmanNode* CreateHuffmanTree(int n, int we[]) {
	// 开辟数组，初始化数组
		// 保存结点权，根据权值来确认每个结点
	HuffmanNode* tree = (HuffmanNode*)malloc(sizeof(HuffmanNode) * (2 * n));
	for (int i = 0; i < n; i++) {
		tree[i].w = we[i];
		tree[i].fa = -1;
		tree[i].lch = -1;
		tree[i].rch = -1;
	}

	// 创建HuffmanTree
	int m1, m2 = -1;	// 保存权值最小的两个结点的下标；	m1 最小 m2 次小

	// 重复n-1次合并
	for (int i = n; i <= 2 * n - 2; i++) {
		Find(tree, i, m1, m2);

		tree[i].fa = -1;
		tree[i].lch = m1;
		tree[i].rch = m2;
		tree[i].w = tree[m1].w + tree[m2].w;

		// 当前结点为父亲结点
		tree[m1].fa = i;
		tree[m2].fa = i;
	}

	return tree;
}

/*
	CreateHuffmanCodes 生成Huffman编码
	构建好HuffmanTree之后，需要通过访问根结点到叶子结点的路径来得到路径编码

	char** CreateHuffmanCodes(HuffmanNode* tree, int n);
	参数：tree 构建好的HuffmanTree； n 要生成编码的叶子结点的个数
	返回值：保存 char链表指针 的数组 char**

	思路：
		父亲结点到子结点，往左是0，往右是1;

		错误：如果从 根结点往 叶子结点走，会发现只能找到其中一个叶子结点，没法继续找了。
		所以需要从叶子结点往根结点走，所以在往数组里装编码时，要从后往前装，因为我们从树上读取的顺序是倒着的，
		这样在存储玩Codes时就会是正着的了。

		然后因为 树是存储在数组中的，且n个结点是存在前n个单元里，
		所以通过数组指针就可以确定 叶子结点的位置。

		因为无法确定每个路径编码的长度，出于节省空间的原则，我们先用足够长的数组保存编码，再拿到合适大小的数组当中去。
		编码最长的长度取决于树的形态，对于n个叶子结点的树，最多能有n-1层，即路径编码长度最多是n-1,所以长度为n-1的数组就足够

		总之，关键在于
			从下往上找路径，保存n个结点的位置，节省空间
*/

char** CreateHuffmanCodes(HuffmanNode* tree, int n) {
	// 暂时存储每个结点的路径编码
	char* t = (char*)malloc(sizeof(char) * n);

	// 二级指针模拟开二维数组，存储char*指针，即n个叶子结点的路径编码
	char** codes = (char**)malloc(sizeof(char*) * n);
	memset(codes, 0, sizeof(char*) * n);

	// 三个指针
	int start;
	int p, pre;

	// 开始倒着访问树并倒着存储
	for (int i = 0; i < n; i++) {
		// 初始化三个指针

		// 指向t数组的最有一个元素，因为倒着读取Code，需要倒着存储，才会在下一次访问时呈现正确的顺序。
		start = n - 1;
		// p指向当前循环的结点的下标
		p = i;
		// pre指向当前结点的父亲结点的下标
		pre = tree[i].fa;

		// 添加字符串字符数组结束标识符'\0'
		t[start] = '\0';


		// 用三个指针进行操作
		while (pre != -1) {
			// 存储指针前移
			start--;

			// 存编码
			if (p == tree[pre].lch) t[start] = '1';
			else t[start] = '0';

			// 树上的指针更新
			p = pre;
			pre = tree[p].fa;
		}
		// 此时start指向最前面一个完成填充的元素

		// 节省空间，迁移t中的字符内容
		codes[i] = (char*)malloc(sizeof(char) * (n - start));
		strcpy(codes[i], &t[start]);	// 从t的start下标元素开始，复制元素到codes[i]指向的地址的空间里去
	}

	return codes;
}


int main() {
	// 对n个字符进行编码，不超过MAXSIZE个
	cout << "请输入要编码的字符的数目: ";
	int n;
	cin >> n;
	while (n > 100) {
		cout << "字符数最多不超过100，请重新输入: ";
		cin >> n;
	}

	// 保存n个字符的内容，后序输出编码时，可以看出是谁的编码
	char c[105];
	cout << "请输入n个字符: ";
	for (int i = 0; i < n; i++) {
		cin >> c[i];
	}

	// 保存n个字符对应的权重
	int we[105];
	cout << "请输入n个字符对应的权重: ";
	for (int i = 0; i < n; i++) {
		cin >> we[i];
	}

	// 利用c和we数组生成HuffmanTree
	HuffmanNode* tree = CreateHuffmanTree(n, we);
	// 利用生成的HuffmanTree来生成HuffmanCodes
		// 利用char**来接收二维数据
	char** codes = CreateHuffmanCodes(tree, n);

	// 遍历输出二维数据,因为有字符结束标识符'\0'，所以直接遍历ans的数据即可，会自动识别每个数组元素的全部内容
	cout << "字符的HuffmanCodes为：" << endl;
	for (int i = 0; i < n; i++) {
		cout << c[i] << " " << codes[i] << endl;
	}

	return 0;
}
