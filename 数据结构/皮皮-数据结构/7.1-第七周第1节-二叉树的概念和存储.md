二叉树：非线性的逻辑结构
	特点:
		每个结点最多有两棵子树，两个后继，不存在度大于2的结点
	有五种形态：
		![[Pasted image 20260202200706.png]]
	具有递归性。
	是有序树。
	特殊的二叉树：
		==满二叉树==：
			定义：
				1、除子结点以外，其余结点的度都为2，且叶子结点都在同一层。
				也就是n层的二叉树最满的状态定义为满
				2、高度为h，且第k(1<=k<=h)层均有2^(k-1)个结点的二叉树
				3、高度为h，且刚好有(2^k)-1个结点的二叉树
		==完全二叉树==：
			定义：
				把满二叉树从上到下 从左到右，从1开始编号，从大编号往小编号连续删除若干个结点，剩下的部分为完全二叉树。（即保证编号从小到大连续）
			性质：
				度为1的结点最多有1个（0或1）
	性质：
		假设二叉树深度为h。根结点在第一层。
			(1)第k(1<=k<=h)层最多有2^(k-1)个结点
			(2)整棵树最多有(2^h)-1个结点(等比数列求和)
			(3)在一棵二叉树中，叶子结点的数目比度为2的结点的数目多一个
				假设叶子结点有n0个，度为2的结点有n2个，一定有==n0=n2+1==
				证明：
					假设一棵二叉树有n个结点：
						n0是叶子结点的个数 n1是度为1的结点的个数 n2是度为2的结点的个数
					n=n0+n1+n2
					n-1=0\*n0+1\*n1+2\*n2---------》分支线思路可得
					证毕
			(4)具有n个结点的完全二叉树的深度是h=floor(logn)+1=ceil(log(n+1))
			floor()向下取整
			ceil()向上取整
			精确计算完全二叉树的高度，粗略计算普通二叉树的高度
			![[Pasted image 20260202210144.png]]
			![[Pasted image 20260202210629.png]]
			(5)对于一棵有n个结点的完全二叉树，按照从上到下 从左往右，从1开始编号。如果一个结点的编号是i 这个结点的父亲是i/2 左孩子2\*i 右孩子2\*i+1

#include <iostream>

/*
	链式存储 二叉树
	思路：
		二叉树最多有两棵子树，左子树/右子树
*/

#define LEFT_CHILD 0
#define RIGHT_CHILD 1

typedef char ElemType;

//  结点结构体
typedef struct Node {
	ElemType data;	// 数据域
	struct Node* lchild;	// 左孩子指针
	struct Node* rchild;	// 右孩子指针
}BTNode, *BTree;

// 初始化链表
BTree InitTree(ElemType data) {
	// 开辟首结点空间
	BTree root = (BTree)malloc(sizeof(BTNode));

	// 初始化赋值
	root->data = data;
	root->lchild = NULL;
	root->rchild = NULL;

	// 返回
	return root;
}

/*
	Find()函数，查找在root树上的数据为x的结点的地址，并返回地址
	BTNode* Find(BTree root, ElemType x);
	参数：
		root 确定是哪棵树
		x 确定要查找的结点
	返回值：
		返回查找到的结点的地址
*/

/*
	思路：
		遍历二叉树
		深度优先遍历 中左右
		递归
*/

BTNode* Find(BTree root, ElemType x) {
	// 递归出口
	if (root == NULL) {
		return NULL;
	}

	// 操作部分
	if (root->data == x) {
		return root;
	}

	BTNode* ans;
	// 递归部分,注意return相当于直接结束本次函数，
	// 在递归部分，只能在最后使用这个，否则会导致部分语句无法执行。
	if (root->lchild != NULL) {
		ans = Find(root->lchild, x);
	}

	if (root->rchild != NULL) {
		ans = Find(root->rchild, x);
	}

	return ans;
}

/*
 二叉树插入结点
	 void Insert(BTree root, ElemType x, ElemType fx, int flag)
	 参数：
			root 要插入到哪棵树中
			x 要插入的结点的数据
			fx 要插入的结点的parent结点的数据，定位到树中的位置
			flag 要插入结点作为parent的左/右子树-----> LEFT_CHILD/RIGHT_CHILD
*/ 

void Insert(BTree root, ElemType x, ElemType fx, int flag) {
	// 创建数据为x的新结点
	BTNode* newNode = (BTNode*)malloc(sizeof(BTNode));
	newNode->data = x;
	newNode->lchild = NULL;
	newNode->rchild = NULL;

	// 找到fx在root上的地址
	BTNode* pos = Find(root, fx);

	// 通过flag确定将x结点插入到哪个位置
	if (flag == LEFT_CHILD && pos->lchild == NULL) {
		pos->lchild = newNode;		
		std::cout << "成功插入结点" << x << "到" << fx << "左子树上!" << std::endl;

	}
	else if (flag == RIGHT_CHILD && pos->rchild == NULL) {
		pos->rchild = newNode;
		std::cout << "成功插入结点" << x << "到" << fx << "右子树上!" << std::endl;
	}
	else {
		std::cout << "插入错误！" << std::endl;
	}

}

/*
* 深度优先遍历 递归版
*	分为 先序遍历 中序遍历 后序遍历
* 这里以先序遍历为例：
*	根（当前结点） 左子树 右子树
*/

void PreOrderBT(BTree root) {
	// 递归出口
	if (root == NULL) {
		return;
	}

	// 访问当前结点数据
	std::cout << root->data << std::endl;

	// 递归到左子树
	PreOrderBT(root->lchild);

	// 递归到右子树
	PreOrderBT(root->rchild);

}

// 中序遍历
void InOrderBT(BTree root) {
	// 递归出口

		return;
	}

	// 递归部分
	
	// 先递归左子树
	InOrderBT(root->lchild);
	// 访问数据

	// 再递归右子树

		return;
	}

	// 递归部分

	// 先递归左子树
	PostOrderBT(root->lchild);
	
	// 再递归右子树

	// 访问数据

// ----------------链栈--------------------

// 链栈结点
typedef struct StackNode {
	// 数据域存放BTNode*
	BTNode* data;
	// next指针成链
	struct StackNode* next;

}StackNode, *Stack;

// 初始化 链栈
Stack InitStack() {
	Stack top = (Stack)malloc(sizeof(StackNode));

	top->data = NULL;
	top->next = NULL;

	return top;
}

// 判空：判断栈内是否有元素
bool isEmpty(Stack top) {

	// 栈为空
	if (top->next == NULL) {
		std::cout << "栈为空" << std::endl;
		return true;
	}
	else {
		//std::cout << "栈不为空" << std::endl;

		// 计数栈中结点个数
		int count = 0;
		StackNode* p = top;
		while (p->next != NULL) {
			count++;
			p = p->next;
		}
		//std::cout << "栈中有" << count << "个元素" << std::endl;

		return false;
	}
}

// 栈中插入结点
void PushStack(Stack top, BTNode* newNode) {
	// 创建新结点并初始化
	StackNode* s = (StackNode*)malloc(sizeof(StackNode));
	s->data = newNode;
	s->next = NULL;

	// 头插法入栈——原因是：栈是同一端进出元素
	s->next = top->next;
	top->next = s;

}

// 栈中弹出结点
// ElemType 说明弹出元素的BTNode指向的结点的数据
ElemType PopStack(Stack top) {
	// 创建指针p指向 头结点的next，即p指向首元结点
	StackNode* p = top->next;

	// 创建对象准备接收结点的data
	// -1表示结点不存在，即栈中没有元素
	ElemType ans = -1;

	// 完成pop操作
	if (p != NULL) {
		top->next = p->next;

		// 接收被pop的结点里的data
		ans = p->data->data;
		
		free(p);
		p = NULL;
	}

	return ans;
}

	BTNode* node = current->data;

	// 访问 树结点
	std::cout << node->data << " ";

	return current;

//=======先序======

/*
思路：
	先把根结点入栈，
	循环中，栈非空时：
		访问 栈顶元素
		出栈 栈顶元素
		入栈 先入栈当前结点的右子树，再入栈当前结点的左子树

	先右后左，原因是 下次循环时，会先出栈左子树来访问，符合先序遍历。
*/
void PreOrderBT(BTree root) {
	// 创建指针p和栈top
	BTNode* p = root;
	Stack top = InitStack();
	
	// 先入栈根结点
	PushStack(top, root);

	// 循环体
		// 当栈非空
	while (!isEmpty(top)) {
		// 找到栈顶元素
		StackNode* current = top->next;

		// 访问栈顶元素
		Visit(current);

		// 出栈
		PopStack(current);

		// 入栈 先右子树，后左子树
		if (current->data->rchild != NULL) {
			PushStack(top, current->data->rchild);
		}
		if (current->data->lchild != NULL) {
			PushStack(top, current->data->lchild);
		}

	}
}

//=======中序========
/*
思路：
	循环体，树结点存在/栈中不存在结点时
		当 树结点存在，
			入栈 树结点，
			根结点指针指向结点左子树
		否则，
			访问栈顶元素，
			出栈栈顶元素，
			根结点指针指向结点右子树
*/
void InOrderBT(BTree root) {
	// 创建指针p指向遍历树上结点
	BTNode* p = root;
	// 创建栈top
	Stack top = InitStack();

	// 循环体
	// 条件：当遍历树仍存在未遍历结点时，或当栈未空时
	while (p != NULL || isEmpty(top)) {
		// 当遍历树仍有未遍历结点时
		if (p != NULL) {
			PushStack(top, p);
			p = p->lchild;
		}
		// 否则
		else {
			StackNode* f = Visit(top);
			PopStack(f);
			// 继续循环
			p = p->rchild;
		}
	}

}
