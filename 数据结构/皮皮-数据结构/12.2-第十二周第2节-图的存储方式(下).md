4、==十字链表(只适用于有向图)==：对有向图的邻接表和逆邻接表合二为一的一种优化。
	对于有向图，邻接表和逆邻接表 并非全能
为了既方便找出度 又方便找入度，可以把邻接表和逆邻接表 合二为一，一个顶点后面挂两个链表
还是有缺陷：（1）浪费空间，一条边对应两个链表结点。（2）操作麻烦

进行优化:一条边只对应一个结点 把同一个结点插入到两个不同的链表中

```
#include<stdio.h>
#include<stdlib.h>
#include<string.h>
#define inf 10001
//带权有向图为例。n个点 m条边  n<100  0<=w<=10000

//十字链表的结点结构
typedef struct ENode
{
	int w;//边权
	int taili;//弧尾下标
	struct ENode* tnext;//指向弧尾下一条出边的指针，通过tnext构成了弧尾的出边链表
	int headi;//弧头下标
	struct ENode* hnext;//指向弧头下一条如边的指针，通过hnext构成了弧头的入边链表
}ENode;
struct Graph
{
	char data;//顶点的数据域
	ENode* firstout;//顶点的出边链表指针
	ENode* firstin;//顶点的入边链表指针
}g[105];
int n,m;
int Find(char x)
{
	int i=0;
	for(i=1;i<=n;i++)
	{
		if(g[i].data==x)
		{
			break;
		}
	}
	return i;
}
int main()
{
	scanf("%d %d",&n,&m);
	getchar();
	for(int i=1;i<=n;i++)
	{
		scanf("%c",&g[i].data);
		g[i].firstin=g[i].firstout=NULL;
	}
	
	char x,y;
	int xi,yi,w;
	for(int i=1;i<=m;i++)
	{
		getchar();
		scanf("%c %c %d",&x,&y,&w);//x->y 有向边
		xi=Find(x);
		yi=Find(y);
		ENode* e=(ENode*)malloc(sizeof(ENode));
		//if(e==NULL)
		e->taili=xi;
		e->headi=yi;
		e->w=w;
		//把xi---》yi 当作xi的出边 插入到xi到出边链表中
		e->tnext=g[xi].firstout;
		g[xi].firstout=e;
	
		//把xi---》yi 当作yi的入边 插入到yi到入边链表中
		e->hnext=g[yi].firstin;
		g[yi].firstin=e;
		
	}
	getchar();
	scanf("%c",&x);
	int outd=0,ind=0;
	xi=Find(x);
	//出度
	ENode* p=g[xi].firstout;
	while(p!=NULL)
	{
		outd++;
		p=p->tnext;
	}
	//入度
	p=g[xi].firstin;
	while(p!=NULL)
	{
		ind++;
		p=p->hnext;
	}
	printf("出度是%d,入度是%d\n",outd,ind);
	return 0;
}
/*
4 5
ABCD
A B 3
A D 6
A C 0
B D 9
D C 4
*/
```



5、==多重邻接表(只适用于无向图)==：无向图的邻接表，本身就是一条边在两个链表中对应两个结点，存两次
	有缺陷：（1）浪费空间，一条边对应两个链表结点。（2）操作麻烦
	仿照有向图的十字链表，进行优化：一条边只对应一个结点 把同一个结点插入到两个不同的链表中
	得到 无向图的多重邻接表。

代码：
```
	#include<stdio.h>
#include<stdlib.h>
#include<string.h>
#define inf 10001
//带权无向图为例。n个点 m条边  n<100  0<=w<=10000
//多重邻接表：边链表的结点结构
typedef struct ENode
{
	int w;//边权
	int x;//端点x的下标
	struct ENode* xnext;//指向x所连接的下一条边的指针，通过xnext构成了的度边链表
	int y;//端点y的下标
	struct ENode* ynext;//指向y下一条边的指针，通过ynext构成了y的边链表
}ENode;
struct Graph
{
	char data;//顶点的数据域
	ENode* first;//顶点的边链表指针
	
}g[105];
int n,m;
int Find(char a)
{
	int i=0;
	for(i=1;i<=n;i++)
	{
		if(g[i].data==a)
		{
			break;
		}
	}
	return i;
}
int main()
{
	scanf("%d %d",&n,&m);
	getchar();
	for(int i=1;i<=n;i++)
	{
		scanf("%c",&g[i].data);
		g[i].first=NULL;
	}
	
	char a,b;
	int ai,bi,w;
	for(int i=1;i<=m;i++)
	{
		getchar();
		scanf("%c %c %d",&a,&b,&w);//x---y 无向边
		ai=Find(a);//找两个端点的编号
		bi=Find(b);

		ENode* e=(ENode*)malloc(sizeof(ENode));
		//if(e==NULL)
		e->x=ai;
		e->y=bi;
		e->w=w;
		//e插入到a的链表中
		e->xnext=g[ai].first;
		g[ai].first=e;
		
		//e插入到b的链表中
		e->ynext=g[bi].first;
		g[bi].first=e;
	}
	getchar();
	scanf("%c",&a);
	int d=0;
	ai=Find(a);
	ENode* p=g[ai].first;
	while(p!=NULL)
	{
		d++;
		if(p->x==ai)p=p->xnext;
		else p=p->ynext;
	}
	printf("度为%d\n",d);
	return 0;
}
/*

ABCD
A B 3
A D 6
A C 0
B D 9
D C 4
*/
```
