基于插入思想的排序：
	直接插入排序 希尔排序

1、默认排成升序
2、假设是对n个数排序，数据用数组a保存
3、大部分算法要把数据分成2块：乱序区 有序区
	一开始：n个位置都是乱序区
	排序的过程中：乱序区变小 有序区变大
	最后：n个位置全是有序区

一、==**直接插入排序**==：
	直观理解：
		![[Pasted image 20260421201005.png]]
	一开始认为有序区是\[1,1]，乱序区是\[2,n]
		进行直接插入排序：
		执行n-1趟排序：每次把乱序区中的第一个数据x 插入到有序区中的对应位置。
			（1）倒着遍历，在有序区中找到第一个小于等于x的位置j，x就应该在j+1位置
			（2）空出j+1位置
			（3）x放到j+1即可
		此时 有序区+1，乱序区-1

对直接插入排序的优化：==折半插入排序==   ==2-路插入排序==

```
#include <iostream>
using namespace std;
#define maxx 100

// 直接插入排序 Straight_Insertion_Sort
// 稳定排序 
// 默认 排成升序
/*
思路：
	int数组 存储数据
	数据区域分为 乱序区 有序区
	排序过程 乱序区逐渐较小 有序区逐渐增大
	初始 默认有序区数据：数组首元素

	保存当前选中的要排序的 数组元素的下标ind 到int k
	for循环 从 ind-1 到 1，a[i] 与 k 比较大小
	if k<a[i], a[i+1]=a[i], 即后移一位
	if k>=a[i], a[i+1]=k;return; 安排好k了，乱序区-1，有序区+1
	更新k值，直到遍历数组
*/
// 参数：数组指针，数组长度
void SISort(int a[],int n)
{
	int k;
	// 当前要加入到有序区的元素的 下标
	for (int i = 2; i <= n; i++)
	{
		k = a[i];
		// 要比较的乱序区的下标范围
		int j;
		for (j = i - 1; j >= 1&& k < a[j]; j--)
			a[j + 1] = a[j];
		a[j + 1] = k;
	}
}

// 读取数组某区域的数据
void readArr(int a[], int l, int r)
{
	for (int i = l; i <= r; i++)
	{
		cout << a[i]<<" ";
	}
}
int main()
{
	int arr[maxx];
	int n; cin >> n;
	for (int i = 1; i <= n; i++)
	{
		cin >> arr[i];
	}

	SISort(arr, n);
	readArr(arr, 1, n);

	return 0;
}
```

二、==**希尔排序**==：shell 排序——**不稳定排序**  **就地排序**
	思路：
		先将所有待排序的数据分组，每组单独进行直接插入排序，使得整体基本有序，最后再对整体执行一次直接插入排序。
	n个数据 假设分成d组 每组n/d个数据
	==d：组数/增量==：同一组中，逻辑上相邻的两个数 下标相差d
		![[Pasted image 20260421220701.png]]
	==时间复杂度==：最好是O(n^1.3) 最坏是O(n^2)

代码：==shell排序==
```
#include<stdio.h>
#include<stdlib.h>
//希尔排序: 时间复杂度最好是O(n^1.3). 最坏是O(n^2)  就地的  不稳定
int n,a[105];
int main()
{
	scanf("%d",&n);
	for(int i=1;i<=n;i++)
	{
		scanf("%d",&a[i]);
	}
	int k=0,x,j;
	for(int d=n/2;d>=1;d=d/2)//枚举增量d(d也是分组的组数)
	{
		k++;//现在是第k趟排序
		for(int i=1+d;i<=n;i++)//从第1组的第二个数开始排 
		{
			//a[i]就是某组中乱序区的第一个数，该组的有序区的最后一个数一定是a[i-d]
			//i所在的组的数据下标：i，i-d,i-2d....
			x=a[i];
			for(j=i-d;j>=1;j=j-d)
			{
				if(a[j]>x)//a[j]大于x 应该移动到本组中的下一个位置
				{
					a[j+d]=a[j];
				}
				else break;

			}
			a[j+d]=x;
		}
		printf("第%d趟排序的结果: ",k);
		for(int i=1;i<=n;i++)
		{
			printf("%d ",a[i]);
		}
		printf("\n");
	}
	

	return 0;
	
}
```
