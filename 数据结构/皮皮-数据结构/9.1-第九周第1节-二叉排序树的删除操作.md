3.删除操作——保证没有重复数据
	在一棵BST中删除数据——》本质是删除数据所在的结点
	（1）查找 找到x所在的结点p
	（2）根据p的度 分三类处理（我们发现，删除后要保证树的各种性质继续成立，这使得对不同度的结点有不懂的处理方式）

```c++
#include <iostream>
using namespace std;
typedef int ElemType;

// 二叉树链式存储
// 结点结构体
typedef struct BSTNode
{
    ElemType data;
    struct BSTNode* l;
    struct BSTNode* r;
}BSTNode,*BSTree;

// 初始化BST
BSTNode* CreateBSTNode(ElemType data)
{
    BSTNode* newNode = (BSTNode*)malloc(sizeof(BSTNode));
    newNode->data = data;
    newNode->l = newNode->r = nullptr;

    return newNode;
}
#define InitBSTree CreateBSTNode// 定义初始化函数

// 插入，关键
// 不断与结点的数据比较大小，小 -》 左子树，大 -》 右子树，循环往复。

// 递归版
BSTNode* Insert(BSTree root, ElemType data)
{
    if (root == nullptr)
    {
        BSTNode* p = CreateBSTNode(data);
        return p;
    }
    if (data > root->data)// 插入的是大数，往右走
        root->r = Insert(root->r, data);
    else// 插入的是小数，往左走，等于也往左走
        root->l = Insert(root->l, data);

    return root;// 这一步是关键，等找到合适的位置插入好之后，需要返回root，才能保持原树结构不变
    // 这一进程返回的root，其实是上一进程的root->l/r，这样就能实现root->l/r=root->l/r;原结构不变
}
// 非递归版
BSTNode* Insert1(BSTree root, ElemType data)
{
    BSTNode* p = root, * pre = nullptr;// p遍历比较，pre记录p上一次指向的位置
    while (p != nullptr)
    {
        pre = p;
        if (data > p->data)
            p = p->r;
        if (data < p->data)
            p = p->l;
    }
    // 此时p==nullptr，pre指向要插入的位置的父亲结点，但还没确定插入的是 左还是右

    // 确定具体位置并插入
    BSTNode* newNode = CreateBSTNode(data);
    if (data > pre->data) pre->r = newNode;
    else pre->l = newNode;// 等于的话也是插入到左边

    return root;
}
// 查找，用二叉排序树进行查找，高效，时间复杂度max O(h),h是树的高度
// 也可以用递归，因为都是检查 root->data 和 data 的大小的操作

// 递归版
// 找不到返回nullptr
BSTNode* Find(BSTree root, ElemType x)
{
    if (root == nullptr||root->data == x )
        return root;
    if (x > root->data)
        return Find(root->r, x);
    else
        return Find(root->l, x);
}
// 非递归版
BSTNode* Find1(BSTree root, ElemType x)
{
    BSTNode* p = root;
    while (p != nullptr&&p->data!=x)
    {
        if (x > p->data) p = p->r;
        else
            p = p->l;
    }
    return p;
}
// 删除
// 递归版
BSTNode* Delete(BSTree root, ElemType x)
{
    if (root == nullptr);
    if (root->data < x) root->r = Delete(root->r, x);
    else if (root->data > x) root->l = Delete(root->l, x);
    else// root->data==x
    {
        // 处理被删除的结点 
        // 分为两种情况:度为 1，度为 2
        // 度为2：被删除结点的 前驱/后继 继承这个结点的位置，然后
        // 删掉 这个 前驱/后继 结点，这时就递归了，转换成了 度为1的情况，递归下去。
        // 度为1：让唯一的孩子继承这个结点的位置
        if (root->l != nullptr && root->r != nullptr)// 度为2时的转换
        {// 这里以 前驱继承 为例，后继思路一样。
            BSTNode* p = root->l;
            while (p->r != nullptr)
            {
                p = p->r;
            }
            // 此时p为删除结点的前驱
            root->data = p->data;// 把前驱数据给被删除结点
            // 然后就是类比这整个函数，Delete()返回的是删除结点后的整棵树，也就是返回的是root
            // 把这棵树返回到整棵树上
            root->l = Delete(root->l, p->data);// 递归下去
            // 情况2 就转换成 情况1 了
        }
        else// 度为1 处理，这时root指向被删除的结点，把root->l/r，也就是还有的那一边，赋给root，这样我们return root时会把左右子树传递回上一个进程，
            // 上一个进程又把返回值传给了上一个进程的root->l/r，这样就实现了跨过被删除的结点，实现该结点的孩子和该结点的父亲的连接。
        {
            BSTNode* t = root;//存取要删除的结点
            if (root->l != nullptr) root = root->l;
            else root = root->r;
            free(t);// 删掉要删除的结点
            t = nullptr;// 临时指针置空
        }
    }
}
// 非递归版
BSTNode* Delete(BSTree root, ElemType x)
{
    if (!root)
    {
        cout << "树为 空树" << endl;
        return root;// 返回null
    }

    // 找到x的位置
    BSTNode* pre, *p = root;
    while (p&&x!=p->data)// 当还没找到时
    {
        pre = p;
        if (x > p->data) p = p->r;
        else p = p->l;
    }
    // p==nullptr停下循环时
    if (!p)
    {
        cout << "以 " << root->data << " 为根的树上没有 " << x << " 数据" << endl;
        return root;// 返回root
    }

    // 找到了x的位置，保存在p上
    
    // 进行删除操作
    // 度为2，需要转换成度为 1/0 的情况
    if (p->l && p->r)
    {
        // 前驱/后继都可以，这里以前驱为例
        BSTNode* t = p->l, * ft=p;// ft初始化为p->l的父亲结点，即p
        while (t->r)// 如果还有右子树
        {
            ft = t;// 一直保存上一个处理的结点，在这里就是 t的父亲结点
            t = t->r;
        }
        // t指向前驱

        // 把前驱的数据data转移到被删除的结点上
        p->data = t->data;

        // 现在要删除的结点变成了 度为0/1 的结点t
        // p始终指向被删除的结点，更新p=t
        p = t;
        // 把 转换后要删除的结点的 父亲结点 保存起来，后面要用
        pre = ft;
    }
    
    // 度为0/1
    // 这里因为是所有的 度为0/1 的结点的删除，不仅仅包括 从度为2转换来的，
    // 所以没法说p只会有左孩子或者没孩子，这里 左右都有可能有孩子
    // 同时也没法说 p是pre的右孩子，这里 p既可以是左孩子，也可以是右孩子
    // 这里p还有可能是root，这样的话，pre就是nullptr了，需要单独处理
    BSTNode* ch;// 接收p的孩子
    if (p->l) ch = p->l;
    else ch = p->r;
    if (pre)
    {
        if (pre->l == p) pre->l = ch;// 如果p是pre的左孩子
        else pre->r = ch;
    }
    else
        root = ch;
    // 记得删掉结点
    free(p);
    p = nullptr;

    return root;
}

int main()
{

    return 0;
}
```
