广度优先遍历（**Breadth First Search**） BFS（基于==队列==）：
	可以根据邻接关系，人为的划分层次，起点上第0层，第0层的邻接的是第1层。。。第i层的邻接的是第i+1层。
	广度优先遍历就类似树的层次遍历。


遍历思路:
	==**记得标记 点 是否入队过，避免重复入队（不同点的邻接点可以是同一个点）**==
	（1）任选一个未被访问的点做起点，起点入队并且打标记。
	（2）循环操作：只要队列非空，队首q出队，访问q，把q的未入队过的邻接的入队并且打标记。
	（3）直到队列为空为止。如果是连通图就结束，如果是非连通图，继续执行（1），直到所有的点被访问一次为止。

1、存图方式：邻接矩阵/邻接表
2、如果是非连通图怎么办？
	只需要在BFS外套一层循环即可，也就是遇到没被访问过的点 就以其为起点执行BFS
3、入队：顶点数据入队 还是把 顶点的下标入队：下标入队

==BFS可以求无权图的最短路径==，**第一次遇到的点的路径长度，就是到该点的最短距离**

#include <iostream>
using namespace std;
#define maxx 105

typedef char ElemType;

/*
广度优先遍历：（邻接表存图 + 用队列处理循环元素）
	邻接表存图：数组包含 图全部结点，每个结点 有自己所有邻接点组成的 链表。
	循环：
		顺序出队队列中的元素，处理元素，将元素邻接点链表中所有的邻接点入队。

	图，从任一点开始，找到该点所有未入队的邻接点，在下一个循环，找到所有邻接点的未入队的邻接点，循环，
	直至找到所有的点（前提是连通）
	对于不连通的图，可以遍历 点数组，以所有的点为起点遍历一遍，之前入队的不会再入队，不会浪费空间，
	这样可以做到遍历图全部的点。
*/
//=================循环队列===============
typedef struct
{
	int data[maxx];
	int front;// 队首指针
	int rear;// 队尾指针
}Queue;
//初始化队列
Queue* initQueue()
{
	Queue* q = (Queue*)malloc(sizeof(Queue));
	for (int i = 0; i < maxx; i++)
		q->data[i] = 0;
	q->front = 0;
	q->rear = 0;

	return q;
}
// 入队
void enQueue(Queue* q, int a)
{
	// 判满
	if ((q->rear + 1) % maxx == q->front)
	{
		cout << "队列已满，无法入队" << endl;
		return;
	}
	q->data[q->rear] = a;
	q->rear = (q->rear + 1) % maxx;
}
bool isEmpty(Queue* q)
{
	if (q->front == q->rear)
		return 1;// 空
	else return 0;// 非空
}
// 出队
int deQueue(Queue* q)
{
	// 判空
	if (isEmpty(q))
	{
		cout << "队空，无法出队" << endl;
		return -1;
	}
	int res = q->data[q->front];
	q->front=(q->front + 1) % maxx;
	return res;
}

// ==============邻接表存无权图==================
int n, m;// n个点，m条边
int vis[maxx];// 标记 点 有没有入队过 没入=0，入过=1
typedef struct ENode
{
	int adj;// 邻接点下标
	struct ENode* next;
}ENode;
struct Graph
{
	ElemType data;
	ENode* first;// ENode链表
}g[maxx];
// 广度优先遍历
void BFS(int p)// 从图中下标为p的点开始 BFS
{
	// 从p开始BFS
	Queue* q = initQueue();
	enQueue(q,p);
	vis[p] = 1;
	int ans;
	//循环------>出队
	while (!isEmpty(q))
	{
		ans = deQueue(q);
		cout << g[ans].data;//访问 点
		ENode* t = g[ans].first;
		while (t != NULL)
		{
			if (vis[t->adj] == 0)//未入队过
			{
				enQueue(q, t->adj);
				vis[t->adj] = 1;//入队时标记状态
			}
			t = t->next;
		}
	}
}
// 通过ElemType数据查找int下标
int Find(ElemType x)
{
	for (int i = 1; i <= n; i++)
	{
		if (g[i].data == x)
		{
			return i;
		}
	}
	return 0;//  表示没有查找到该数据
}
int main()
{
	cin >> n >> m;// 点数 边数
	// 输入n个点到Graph g[maxx]中
	ElemType a; 
	for (int i = 1; i <= n; i++)
	{
		cin >> a;// 点
		g[i].data = a;
		g[i].first = NULL;
	}
	// 输入n个边到邻接表中，这是无向图
	ElemType x, y;
	int xi, yi;
	for (int i = 1; i <= m; i++)
	{
		cin >> x >> y;// 边
		xi = Find(x);
		yi = Find(y);

		// y做x的邻接点
		ENode* e = (ENode*)malloc(sizeof(ENode));
		e->adj = yi;
		e->next = g[xi].first;
		g[xi].first = e;

		// x做y的邻接点
		e = (ENode*)malloc(sizeof(ENode));
		e->adj = xi;
		e->next = g[yi].first;
		g[yi].first = e;
	}
	for (int i = 1; i <= n; i++)
	{
		if(vis[i]==0)
			BFS(i);
	}

	return 0;
}
